// ============== CONVERSÃO NUMÉRICA ==============
/**
 * Converte string brasileira para número.
 * "360.530,60" -> 360530.60
 * "360.530"    -> 360530
 * "360,60"     -> 360.60
 * 360530       -> 360530 (já é número, retorna direto)
 * ""           -> "" (vazio retorna vazio)
 */
function _toNum(v) {
  if (v === '' || v === null || v === undefined) return '';
  if (typeof v === 'number') return v;
  var s = String(v).trim();
  if (s === '') return '';
  // Remove pontos de milhar e troca vírgula decimal por ponto
  var normalizado = s.replace(/\./g, '').replace(',', '.');
  var n = Number(normalizado);
  return isNaN(n) ? v : n;
}

/**
 * Aplica _toNum em todos os campos numéricos de um row.
 */
function _normalizarNumericos(row) {
  if (!row) return row;
  CAMPOS_NUMERICOS.forEach(function(campo) {
    if (Object.prototype.hasOwnProperty.call(row, campo)) {
      row[campo] = _toNum(row[campo]);
    }
  });
  return row;
}

// ============== LEITURA ==============
function _readSheet(name) {
  var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(name);
  if (!sh) return { headers: [], rows: [] };
  var values = sh.getDataRange().getValues();
  if (!values.length) return { headers: [], rows: [] };

  var headers = values[0].map(function(h){ return String(h||'').trim(); });
  var rows = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    var hasAny = false;
    for (var k=0; k<row.length; k++) {
      if (row[k] !== '' && row[k] !== null && row[k] !== undefined) { hasAny = true; break; }
    }
    if (!hasAny) continue;

    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = _formatSheetDate(row[j], headers[j]);
    }
    rows.push(obj);
  }
  return { headers: headers, rows: rows };
}

// ============== FORMATAÇÃO DE CÉLULA ==============
function _formatSheetDate(val, headerName) {
  if (val === '' || val === null || val === undefined) return '';

  var h = String(headerName||'').toUpperCase();
  var isTimeCol = (h.indexOf('HORA') >= 0 ||
                   h.indexOf('CHEGADA') >= 0 ||
                   h.indexOf('SAIDA') >= 0 ||
                   h.indexOf('SAÍDA') >= 0 ||
                   h.indexOf('INICIO') >= 0 ||
                   h.indexOf('INÍCIO') >= 0 ||
                   h.indexOf('FIM') >= 0);

  if (Object.prototype.toString.call(val) === '[object Date]' && !isNaN(val)) {
    var y = val.getFullYear();
    if (y === 1899 && val.getMonth() === 11 && val.getDate() === 30) {
      return Utilities.formatDate(val, TZ, 'HH:mm');
    }
    if (isTimeCol) {
      return Utilities.formatDate(val, TZ, 'HH:mm');
    }
    return Utilities.formatDate(val, TZ, 'yyyy-MM-dd');
  }

  if (typeof val === 'number') return val;
  if (typeof val === 'boolean') return val ? 'TRUE' : 'FALSE';

  var isNumericCol = (h.indexOf('QUANTIDADE') >= 0 ||
                      h.indexOf('LITROS') >= 0 ||
                      h.indexOf('VALOR') >= 0 ||
                      h === 'KM' ||
                      h.indexOf('INTERVALO KM') >= 0 ||
                      h.indexOf('ALERTA') >= 0 ||
                      h === 'LATITUDE' || h === 'LONGITUDE');
  if (isNumericCol) {
    var s = String(val).trim();
    if (s !== '') {
      var normalizado;
      if (s.indexOf(',') >= 0) {
        normalizado = s.replace(/\./g, '').replace(',', '.');
      } else {
        normalizado = s;
      }
      var n = Number(normalizado);
      if (!isNaN(n)) return n;
    }
  }

  return String(val);
}

// ============== UTILITÁRIOS DE CHAVE ==============
function _normalizeKey(s) {
  return String(s||'').trim().toUpperCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g,'');
}

function _extractDatePart(v) {
  if (v === '' || v === null || v === undefined) return '';
  if (Object.prototype.toString.call(v) === '[object Date]' && !isNaN(v)) {
    return Utilities.formatDate(v, TZ, 'yyyy-MM-dd');
  }
  var s = String(v).trim();
  var m = s.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (m) return m[1]+'-'+m[2]+'-'+m[3];
  m = s.match(/^(\d{2})\/(\d{2})\/(\d{4})/);
  if (m) return m[3]+'-'+m[2]+'-'+m[1];
  return s;
}

function _sameDayRegistro(a, b) {
  return _extractDatePart(a) === _extractDatePart(b);
}

function _matchesKey(rowObj, key) {
  var rMot = _normalizeKey(rowObj['MOTORISTA']);
  var rDt  = _extractDatePart(rowObj['DATA']);
  var rReg = _extractDatePart(rowObj['DATA_REGISTRO'] || rowObj['DATAREGISTRO'] || '');

  var kMot = _normalizeKey(key.motorista);
  var kDt  = _extractDatePart(key.data);
  var kReg = _extractDatePart(key.dataRegistro);

  if (rMot !== kMot) return false;
  if (rDt  !== kDt)  return false;
  if (rReg && kReg && rReg !== kReg) return false;
  return true;
}

// ============== BUSCA DE LINHA NA CADASTRO ==============
function _findCadastroRow(body) {
  var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_CADASTRO);
  if (!sh) throw new Error('Aba CADASTRO não encontrada');

  var values = sh.getDataRange().getValues();
  if (values.length < 2) return null;

  var headers = values[0].map(function(h){ return String(h||'').trim(); });
  var idxId = headers.indexOf('ID');

  if (body.id !== undefined && body.id !== null && body.id !== '' && idxId >= 0) {
    var alvoId = String(body.id).trim();
    for (var i = 1; i < values.length; i++) {
      var cur = String(values[i][idxId]||'').trim();
      if (cur && cur === alvoId) {
        return { sheet: sh, rowIndex: i+1, headers: headers, currentValues: values[i] };
      }
    }
    return null;
  }

  var key = {
    motorista:    body.motorista    || body.keyMotorista    || (body.row && body.row.MOTORISTA),
    data:         body.data         || body.keyData         || (body.row && body.row.DATA),
    dataRegistro: body.dataRegistro || body.keyDataRegistro || (body.row && body.row.DATA_REGISTRO)
  };
  for (var j = 1; j < values.length; j++) {
    var obj = {};
    for (var k = 0; k < headers.length; k++) obj[headers[k]] = values[j][k];
    if (_matchesKey(obj, key)) {
      return { sheet: sh, rowIndex: j+1, headers: headers, currentValues: values[j] };
    }
  }
  return null;
}

// ============== ID AUTOMÁTICO ==============
function _proximoId(sheet, headers) {
  var idxId = headers.indexOf('ID');
  if (idxId < 0) return 1;
  var lastRow = sheet.getLastRow();
  if (lastRow < 2) return 1;
  var col = sheet.getRange(2, idxId+1, lastRow-1, 1).getValues();
  var maxId = 0;
  for (var i = 0; i < col.length; i++) {
    var n = parseInt(col[i][0], 10);
    if (!isNaN(n) && n > maxId) maxId = n;
  }
  return maxId + 1;
}

// ============== GERAR IDs RETROATIVOS ==============
function gerarIDsRetroativos() {
  var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_CADASTRO);
  if (!sh) throw new Error('Aba CADASTRO não encontrada');
  var values = sh.getDataRange().getValues();
  if (values.length < 2) {
    Logger.log('CADASTRO vazia, nada a fazer.');
    return;
  }
  var headers = values[0].map(function(h){ return String(h||'').trim(); });
  var idxId  = headers.indexOf('ID');
  if (idxId < 0) {
    throw new Error('Coluna ID não encontrada. Crie a coluna "ID" como primeira coluna (A) e rode de novo.');
  }
  var idxData = headers.indexOf('DATA');
  var idxReg  = headers.indexOf('DATA_REGISTRO');

  var maxId = 0;
  var jaTem = [];
  var paraPreencher = [];
  for (var i = 1; i < values.length; i++) {
    var raw = values[i][idxId];
    var n = parseInt(raw, 10);
    if (!isNaN(n) && n > 0) {
      jaTem.push(i);
      if (n > maxId) maxId = n;
    } else {
      paraPreencher.push(i);
    }
  }

  paraPreencher.sort(function(a, b){
    var ra = _extractDatePart(idxReg >= 0 ? values[a][idxReg] : '') || _extractDatePart(idxData >= 0 ? values[a][idxData] : '');
    var rb = _extractDatePart(idxReg >= 0 ? values[b][idxReg] : '') || _extractDatePart(idxData >= 0 ? values[b][idxData] : '');
    if (!ra && !rb) return a - b;
    if (!ra) return 1;
    if (!rb) return -1;
    if (ra < rb) return -1;
    if (ra > rb) return 1;
    return a - b;
  });

  var prox = maxId + 1;
  var atualizacoes = [];
  for (var p = 0; p < paraPreencher.length; p++) {
    var lin = paraPreencher[p];
    atualizacoes.push({ row: lin+1, id: prox });
    prox++;
  }

  if (atualizacoes.length === 0) {
    Logger.log('Nenhuma linha precisava de ID. Já tinha %s linhas com ID.', jaTem.length);
    return;
  }
  for (var u = 0; u < atualizacoes.length; u++) {
    sh.getRange(atualizacoes[u].row, idxId+1).setValue(atualizacoes[u].id);
  }
  Logger.log('IDs preenchidos: %s linhas. Já tinham ID: %s. Maior ID agora: %s.',
             atualizacoes.length, jaTem.length, prox - 1);
}

// ============== REPLICAÇÃO NA BASE DE DADOS ==============
function _appendNaBase(nomeColuna, valor) {
  if (valor === '' || valor === null || valor === undefined) return;
  var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_BASE);
  if (!sh) {
    Logger.log('AVISO: aba BASE DE DADOS não encontrada para replicar ' + nomeColuna);
    return;
  }
  var headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
    .map(function(h){ return String(h||'').trim(); });
  var idxCol = headers.indexOf(nomeColuna);
  if (idxCol < 0) {
    Logger.log('AVISO: coluna "' + nomeColuna + '" não encontrada na BASE DE DADOS');
    return;
  }

  var lastRow = sh.getLastRow();
  var valorNorm = _normalizeKey(valor);

  var primeiraVazia = -1;
  if (lastRow >= 2) {
    var col = sh.getRange(2, idxCol+1, lastRow-1, 1).getValues();
    for (var i = 0; i < col.length; i++) {
      var celula = col[i][0];
      if (celula === '' || celula === null || celula === undefined) {
        if (primeiraVazia < 0) primeiraVazia = i + 2;
      } else if (_normalizeKey(celula) === valorNorm) {
        Logger.log('Valor "' + valor + '" já existe em ' + nomeColuna + ', pulando.');
        return;
      }
    }
  }

  var linhaDestino = (primeiraVazia > 0) ? primeiraVazia : (lastRow + 1);
  sh.getRange(linhaDestino, idxCol+1).setValue(valor);
  Logger.log('Replicado "' + valor + '" em ' + nomeColuna + ' (linha ' + linhaDestino + ')');
}

function sincronizarBaseDados() {
  Logger.log('=== INICIANDO SINCRONIZAÇÃO COM BASE DE DADOS ===');

  var shMot = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_MOTORISTAS);
  if (shMot) {
    var valsMot = shMot.getDataRange().getValues();
    var headersMot = valsMot[0].map(function(h){ return String(h||'').trim(); });
    var idxNomeMot = headersMot.indexOf('NOME');
    if (idxNomeMot < 0) idxNomeMot = headersMot.indexOf('MOTORISTA');
    if (idxNomeMot >= 0) {
      Logger.log('-- Replicando motoristas --');
      for (var i = 1; i < valsMot.length; i++) {
        var nome = valsMot[i][idxNomeMot];
        if (nome) _appendNaBase(BASE_COL_MOTORISTA, nome);
      }
    }
  }

  var shCam = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_CAMINHOES);
  if (shCam) {
    var valsCam = shCam.getDataRange().getValues();
    var headersCam = valsCam[0].map(function(h){ return String(h||'').trim(); });
    var idxPlaca = headersCam.indexOf('PLACA');
    if (idxPlaca >= 0) {
      Logger.log('-- Replicando placas --');
      for (var j = 1; j < valsCam.length; j++) {
        var placa = valsCam[j][idxPlaca];
        if (placa) _appendNaBase(BASE_COL_PLACA, placa);
      }
    }
  }

  var shLoc = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_LOCAIS);
  if (shLoc) {
    var valsLoc = shLoc.getDataRange().getValues();
    var headersLoc = valsLoc[0].map(function(h){ return String(h||'').trim(); });
    var idxNomeLoc = headersLoc.indexOf('NOME');
    var idxTipoLoc = headersLoc.indexOf('TIPO');
    if (idxNomeLoc >= 0 && idxTipoLoc >= 0) {
      Logger.log('-- Replicando locais --');
      for (var k = 1; k < valsLoc.length; k++) {
        var nomeLoc = valsLoc[k][idxNomeLoc];
        var tipoLoc = valsLoc[k][idxTipoLoc];
        if (nomeLoc && tipoLoc) {
          _replicarLocalPorTipo(nomeLoc, tipoLoc);
        }
      }
    }
  }

  Logger.log('=== SINCRONIZAÇÃO CONCLUÍDA ===');
}

function _replicarLocalPorTipo(nome, tipo) {
  var t = _normalizeKey(tipo);
  if (t === 'CARGA') {
    _appendNaBase(BASE_COL_LOCAL_CARGA, nome);
  } else if (t === 'DESCARGA') {
    _appendNaBase(BASE_COL_LOCAL_DESCARGA, nome);
  } else if (t === 'ABASTECIMENTO') {
    _appendNaBase(BASE_COL_LOCAL_ABASTECIMENTO, nome);
  } else if (t === 'DESPESA') {
    Logger.log('Local "' + nome + '" do tipo Despesa não replicado (sem coluna na BASE)');
  } else {
    Logger.log('AVISO: tipo desconhecido "' + tipo + '" para local "' + nome + '"');
  }
}

// ============== CADASTRO (CRUD) ==============
function addCadastro(row) {
  if (!row) return { ok:false, error:'row vazio' };
  var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_CADASTRO);
  if (!sh) return { ok:false, error:'Aba CADASTRO não encontrada' };

  // Normalizar campos numéricos antes de salvar
  row = _normalizarNumericos(row);

  var headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]
    .map(function(h){ return String(h||'').trim(); });

  var idxId = headers.indexOf('ID');
  var novoId = null;
  if (idxId >= 0 && (row.ID === undefined || row.ID === null || row.ID === '')) {
    novoId = _proximoId(sh, headers);
    row.ID = novoId;
  }

  var linha = headers.map(function(h){
    var v = row[h];
    return (v === undefined || v === null) ? '' : v;
  });
  sh.appendRow(linha);
  return { ok:true, id: novoId, row: row };
}

function updateCadastro(body) {
  var found = _findCadastroRow(body);
  if (!found) return { ok:false, error:'Linha não encontrada' };

  var row = body.row || body.updates || {};

  // Normalizar campos numéricos antes de salvar

  var headers = values[0].map(function(h){ return String(h||'').trim(); });
  var idxId = headers.indexOf('ID');

  var rowIndex = -1;

  if (idxId >= 0 && body.id !== undefined && body.id !== null && body.id !== '') {
    var alvo = String(body.id).trim();
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][idxId]||'').trim() === alvo) { rowIndex = i+1; break; }

      if (okP && okD && okT) { rowIndex = j+1; break; }
    }
  }

  if (rowIndex < 0) return { ok:false, error:'Linha não encontrada' };

  var rowData = body.row || {};
  var novaLinha = headers.map(function(h, i){
    if (Object.prototype.hasOwnProperty.call(rowData, h)) {
      var v = rowData[h];
      return (v === undefined || v === null) ? '' : v;
    }
    return values[rowIndex-1][i];

  return { ok:true };
}

function deleteTanqueEntrada(body) { return _deleteMaqById(ABA_TANQUE_ENTRADAS, body); }

// Grava o mês inteiro de uma vez (lote). Recebe body.marcacoes = [{DATA, MOTORISTA, CODIGO}, ...]
// Faz UPSERT por MOTORISTA+DATA: se já existe a linha daquele motorista/dia, atualiza o CODIGO;
// senão, cria com ID novo. Assim salvar o mesmo mês de novo não duplica linhas.
// CODIGO vazio/null => remove a marcação daquele dia (se existir).
function saveFrequenciaMes(body) {
  var marcacoes = (body && body.marcacoes) || [];
  if (!marcacoes.length) return { ok:true, gravadas:0, atualizadas:0, removidas:0, info:'nada a gravar' };

  var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_FREQUENCIA);
  if (!sh) return { ok:false, error:'Aba não encontrada: ' + ABA_FREQUENCIA };

  var values = sh.getDataRange().getValues();
  var headers = (values[0] || []).map(function(h){ return String(h||'').trim(); });
  var idxId   = headers.indexOf('ID');
  var idxData = headers.indexOf('DATA');
  var idxMot  = headers.indexOf('MOTORISTA');
  var idxCod  = headers.indexOf('CODIGO');
  var idxUsu  = headers.indexOf('USUARIO');
  var idxReg  = headers.indexOf('DATA_REGISTRO');
  if (idxData < 0 || idxMot < 0 || idxCod < 0) {
    return { ok:false, error:'Cabeçalho da aba FREQUENCIA precisa ter DATA, MOTORISTA e CODIGO' };
  }

  // Índice das linhas existentes por chave MOTORISTA||DATA (normalizada)
  var mapaLinha = {}; // chave -> número da linha na planilha (1-based)
  for (var i = 1; i < values.length; i++) {
    var dataP = _extractDatePart(values[i][idxData]);
    var motN  = _normalizeKey(values[i][idxMot]);
    if (!dataP && !motN) continue;
    mapaLinha[motN + '||' + dataP] = i + 1;
  }

  var usuario = (body && body.usuario) || '';
  var hoje = _hojeISO();
  var maxId = _proximoId(sh, headers) - 1; // último ID usado
  var gravadas = 0, atualizadas = 0, removidas = 0;

  for (var m = 0; m < marcacoes.length; m++) {
    var mc = marcacoes[m] || {};
    var dataP = _extractDatePart(mc.DATA);
    var motN  = _normalizeKey(mc.MOTORISTA);
    var cod   = String(mc.CODIGO == null ? '' : mc.CODIGO).trim();
    if (!dataP || !motN) continue; // linha incompleta: ignora, não quebra

    var chave = motN + '||' + dataP;
    var linhaExistente = mapaLinha[chave];

    if (cod === '') {
      // remoção: se existe, apaga a linha
      if (linhaExistente) {
        sh.deleteRow(linhaExistente);
        removidas++;
        // reindexar mapa (linhas abaixo subiram 1)
        for (var key in mapaLinha) {
          if (mapaLinha[key] > linhaExistente) mapaLinha[key] = mapaLinha[key] - 1;
        }
        delete mapaLinha[chave];
        values.splice(linhaExistente - 1, 1);
      }
      continue;
    }

    if (linhaExistente) {
      // atualiza só o CODIGO (e USUARIO/DATA_REGISTRO da alteração)
      if (idxCod >= 0) sh.getRange(linhaExistente, idxCod+1).setValue(cod);
      if (idxUsu >= 0 && usuario) sh.getRange(linhaExistente, idxUsu+1).setValue(usuario);
      if (idxReg >= 0) sh.getRange(linhaExistente, idxReg+1).setValue(hoje);
      atualizadas++;
    } else {
      // cria linha nova com ID
      maxId++;
      var nova = headers.map(function(h){
        if (h === 'ID') return maxId;
        if (h === 'DATA') return dataP;
        if (h === 'MOTORISTA') return mc.MOTORISTA;
        if (h === 'CODIGO') return cod;
        if (h === 'USUARIO') return usuario;
        if (h === 'DATA_REGISTRO') return hoje;
        return '';
      });
      sh.appendRow(nova);
      mapaLinha[chave] = sh.getLastRow();
      gravadas++;
    }
  }

  return { ok:true, gravadas:gravadas, atualizadas:atualizadas, removidas:removidas };
}

// Atualiza um motorista existente na aba MOTORISTAS, casando por NOME (a aba não tem ID).
// Atualiza SOMENTE os campos enviados em body.row; preserva os demais (CPF/CNH/etc).
// Usado principalmente para gravar DATA_DESLIGAMENTO (e reativar = enviar DATA_DESLIGAMENTO vazio).
function updateMotorista(body) {
  var sh = SpreadsheetApp.openById(SHEET_ID).getSheetByName(ABA_MOTORISTAS);
  if (!sh) return { ok:false, error:'Aba não encontrada: ' + ABA_MOTORISTAS };
  var values = sh.getDataRange().getValues();
  if (values.length < 2) return { ok:false, error:'Aba MOTORISTAS vazia' };
  var headers = values[0].map(function(h){ return String(h||'').trim(); });
  var idxNome = headers.indexOf('NOME');
  if (idxNome < 0) idxNome = headers.indexOf('MOTORISTA');
  if (idxNome < 0) return { ok:false, error:'Coluna NOME não encontrada em MOTORISTAS' };

  var alvoNome = (body && (body.nome || body.NOME)) || (body && body.row && (body.row.NOME || body.row.nome)) || '';
  alvoNome = _normalizeKey(alvoNome);
  if (!alvoNome) return { ok:false, error:'NOME obrigatório para editar motorista' };

  var rowIndex = -1;
  for (var i = 1; i < values.length; i++) {
    if (_normalizeKey(values[i][idxNome]) === alvoNome) { rowIndex = i + 1; break; }
  }
  if (rowIndex < 0) return { ok:false, error:'Motorista não encontrado na aba MOTORISTAS: ' + (body.nome || body.NOME || '') };

  var rowData = (body && body.row) || {};
  var novaLinha = headers.map(function(h, i){
    if (Object.prototype.hasOwnProperty.call(rowData, h)) {
      var v = rowData[h];
      return (v === undefined || v === null) ? '' : v;
    }
    return values[rowIndex-1][i]; // preserva o que não veio
  });
  sh.getRange(rowIndex, 1, 1, headers.length).setValues([novaLinha]);
  return { ok:true, nome: (body.nome || body.NOME || '') };
}
